import sys
line=int(input())
flowers=list(map(int,input().split()))

honeys=0
flowers_copy=flowers.copy()
total=0
for i in range(line):
    for j in range(line):
        if j == i:
            flowers_copy=flowers.copy()
            continue
        
        for k in range(line):
            flowers_copy[i]=0
            flowers_copy[j]=0
            flowers_copy[k]=0
            if k == i or k == j:
                flowers_copy=flowers.copy()
                continue
            if i < j:
                total+=sum(flowers_copy[i:j]+1)
            else:
                total+=sum(flowers_copy[j:i]+1)
            
            if i < k:
                total+=sum(flowers_copy[i:k]+1)
            else:
                total+=sum(flowers_copy[k:i]+1)
            if total > honeys:
                honeys=total
            flowers_copy=flowers.copy()
            total=0

print(honeys)
